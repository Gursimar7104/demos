Read-only MCP health check for a broken Ubuntu-style box

3 of 3 server faults caught by a read-only MCP check, and 3 of 3 cleared by heal.sh

Built by Gursimar Singh, Nov 2026

Your post describes a box with a service that won't stay up, a nightly report that stopped, and a disk that keeps filling. I built my own small lab with those three fault types (not your opsbox challenge, since you asked that solutions stay private), a TypeScript MCP server an agent can call to check on it without being able to change anything, and a heal script. The check flagged all 3 faults, and it also caught my first heal.sh: it reported success while the disk was still 100% full, so 2 of 3 checks still failed until I fixed it.

The MCP server (src/server.ts)

// Read-only MCP server: lets an AI agent check a box's health without being able to change it.
// No tool writes files, kills processes or runs shell commands; every path is fixed under LAB_ROOT.
import { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio.js";
import { z } from "zod";
import { promises as fs } from "node:fs";
import path from "node:path";

const ROOT = process.env.LAB_ROOT ?? "/";
const p = (...s: string[]) => path.join(ROOT, ...s);
const SERVICES: Record<string, { pidfile: string; log: string }> = {
  "app-sync": { pidfile: p("run/app-sync.pid"), log: p("var/log/app-sync.log") },
};
const DISK_LIMIT_PCT = 90, REPORT_MAX_AGE_H = 26, MIN_UPTIME_S = 30;
const ok = (o: unknown) => ({ content: [{ type: "text" as const, text: JSON.stringify(o, null, 2) }] });

async function tail(file: string, n: number) {
  try { return (await fs.readFile(file, "utf8")).trimEnd().split("\n").slice(-n); } catch { return []; }
}
async function uptimeSeconds(pid: number) {
  // field 22 of /proc/<pid>/stat is start time in clock ticks since boot
  const stat = await fs.readFile(`/proc/${pid}/stat`, "utf8");
  const start = Number(stat.slice(stat.lastIndexOf(")") + 2).split(" ")[19]) / 100;
  const up = Number((await fs.readFile("/proc/uptime", "utf8")).split(" ")[0]);
  return Math.round(up - start);
}

export async function serviceStatus(name: string) {
  const s = SERVICES[name];
  let pid: number | null = null, alive = false, uptime = 0;
  try { pid = Number((await fs.readFile(s.pidfile, "utf8")).trim()); uptime = await uptimeSeconds(pid); alive = true; } catch {}
  const since = Date.now() - 10 * 60_000;
  const exits = (await tail(s.log, 2000)).filter(l => l.includes("exited code=") && Date.parse(l.split(" ")[0]) >= since);
  const healthy = alive && uptime >= MIN_UPTIME_S;
  return { check: "service_status", service: name, healthy, pid, alive, uptime_s: uptime,
    restarts_last_10m: exits.length, last_exit: exits.at(-1) ?? null,
    verdict: healthy ? "running and stable" : exits.length >= 3 ? "crash-looping" : "not running" };
}
export async function diskUsage() {
  const mount = p("data"); const st = await fs.statfs(mount);
  const pct = Math.round(100 * (1 - st.bavail / st.blocks));
  const files: { file: string; mb: number }[] = [];
  async function walk(d: string) {
    for (const e of await fs.readdir(d, { withFileTypes: true })) {
      const f = path.join(d, e.name);
      if (e.isDirectory()) await walk(f); else files.push({ file: path.relative(ROOT, f), mb: +((await fs.stat(f)).size / 1048576).toFixed(1) });
    }
  }
  await walk(mount); files.sort((a, b) => b.mb - a.mb);
  return { check: "disk_usage", mount: path.relative(ROOT, mount), healthy: pct < DISK_LIMIT_PCT, used_pct: pct,
    size_mb: Math.round(st.blocks * st.bsize / 1048576), limit_pct: DISK_LIMIT_PCT, largest_files: files.slice(0, 3) };
}
export async function lastReportAge() {
  const dir = p("data/reports"); let newest: { file: string; mtime: number } | null = null;
  for (const f of await fs.readdir(dir)) {
    if (!/^report-.*\.txt$/.test(f)) continue;
    const m = (await fs.stat(path.join(dir, f))).mtimeMs; if (!newest || m > newest.mtime) newest = { file: f, mtime: m };
  }
  const age = newest ? +((Date.now() - newest.mtime) / 3.6e6).toFixed(1) : null;
  return { check: "last_report_age", healthy: age !== null && age <= REPORT_MAX_AGE_H, newest_report: newest?.file ?? null,
    age_hours: age, max_age_hours: REPORT_MAX_AGE_H, last_job_log: (await tail(p("var/log/report.log"), 1))[0] ?? null };
}
export async function recentErrors(lines: number) {
  const out: string[] = [];
  for (const f of await fs.readdir(p("var/log"))) for (const l of await tail(p("var/log", f), 500)) if (/ERROR/.test(l)) out.push(`${f}: ${l}`);
  return { check: "recent_errors", count: out.length, errors: out.slice(-lines) };
}

const server = new McpServer({ name: "box-health", version: "0.1.0" });
server.registerTool("service_status", { description: "Is a service up and stable? Reports pid, uptime and restarts in the last 10 minutes.", inputSchema: { name: z.enum(["app-sync"]) } }, async ({ name }) => ok(await serviceStatus(name)));
server.registerTool("disk_usage", { description: "Used % of the data disk and its 3 largest files.", inputSchema: {} }, async () => ok(await diskUsage()));
server.registerTool("last_report_age", { description: "Age in hours of the newest nightly report, plus the job's last log line.", inputSchema: {} }, async () => ok(await lastReportAge()));
server.registerTool("recent_errors", { description: "Last ERROR lines across the service and job logs.", inputSchema: { lines: z.number().int().min(1).max(50).default(10) } }, async ({ lines }) => ok(await recentErrors(lines)));
server.registerTool("health_summary", { description: "Runs service, disk and report checks and counts faults.", inputSchema: {} }, async () => {
  const checks = [await serviceStatus("app-sync"), await diskUsage(), await lastReportAge()];
  return ok({ faults: checks.filter(c => !c.healthy).length, of: checks.length, checks });
});
await server.connect(new StdioServerTransport());

heal.sh

#!/usr/bin/env bash
# self-heal: safe to run repeatedly; each step checks before it acts and verifies after
set -u; export LAB_ROOT="${LAB_ROOT:-$(cd "$(dirname "$0")" && pwd)/box}"; H="$(cd "$(dirname "$0")" && pwd)"
D="$LAB_ROOT/data"; say(){ echo "$(date -u +%T) $*"; }
# 1. disk: stop the runaway writer, keep the last 1 MB of its log for debugging
if [ -f "$LAB_ROOT/run/debug-writer.pid" ] && kill -0 "$(cat "$LAB_ROOT/run/debug-writer.pid")" 2>/dev/null; then
  kill "$(cat "$LAB_ROOT/run/debug-writer.pid")"; say "stopped runaway debug writer"; fi
pkill -f "$H/debug-writer.sh" 2>/dev/null
# v2: a temp file on the full disk can't be written (v1 bug), so stage the tail off-disk and truncate in place
rm -f "$D/logs/debug.log.tail"
if [ -f "$D/logs/debug.log" ]; then T=$(mktemp); tail -c 1048576 "$D/logs/debug.log" > "$T" && truncate -s 0 "$D/logs/debug.log" && cat "$T" >> "$D/logs/debug.log" && say "trimmed debug.log to 1 MB" || say "FAILED to trim debug.log"; rm -f "$T"; fi
say "data disk now $(df --output=pcent "$D" | tail -1 | tr -d ' ')"
# 2. service: restore missing config from the shipped example; supervisor restarts it
CFG="$LAB_ROOT/etc/app-sync/config.json"
if [ ! -s "$CFG" ]; then cp "$LAB_ROOT/etc/app-sync/config.example.json" "$CFG"; say "restored $CFG from example"; fi
pgrep -f "$H/supervisor.sh" >/dev/null || { nohup "$H/supervisor.sh" >/dev/null 2>&1 & say "started supervisor"; }
# 3. report: rerun tonight's job now that there is space
"$H/report.sh" && say "report re-run OK" || say "report re-run FAILED"
sleep 35; P=$(cat "$LAB_ROOT/run/app-sync.pid"); kill -0 "$P" 2>/dev/null && say "app-sync pid $P still up after 35s" || say "app-sync NOT stable"

How it works

  • Lab on a Linux box: a 64 MB tmpfs as the data disk, a Node service under a restart-always supervisor with its config missing, a runaway debug log, and a report job whose last good run was 4 days ago. Tonight's run fails because the disk is full.
  • The MCP server (TypeScript, official SDK, stdio) exposes service_status, disk_usage, last_report_age, recent_errors and health_summary. No tool writes, kills or runs a shell, and paths are fixed, so an agent can look but never make it worse.
  • heal.sh stops the writer, trims the log in place, restores the config and reruns the report, checking each step. An MCP client then reruns every check, which is how the v1 full-disk bug got caught.

What I'd improve next: Point the same tools at systemd and journald on a real Ubuntu 24.04 box (systemctl show NRestarts, journalctl -p err), add logrotate and a disk alert before 90%, and register the server in Claude Code and Copilot Studio so a client's IT person can just ask whether the box is healthy.
